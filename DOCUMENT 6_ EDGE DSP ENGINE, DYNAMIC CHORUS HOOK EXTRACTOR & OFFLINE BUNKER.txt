export interface AcousticTelemetryFrame {
  snrDb: number;
  fundamentalFreqHz: number;
  pitchVariance: number;
  zcrVariance: number;
  classification: "SILENCE" | "ROOM_BANTER_SHIELDED" | "INTENTIONAL_BARGE_IN";
}


export interface ChorusWindowResult {
  startSec: number;
  endSec: number;
  peakRmsEnergy: number;
  vocalPresenceRatio: number;
}


/**
 * 1. ACOUSTIC BANTER SHIELD & BARGE-IN DETECTOR (SUB-ENGINE 3)
 * Separates multi-person laughter/chatter from a sharp single-user interruption.
 */
export class AcousticBanterAndBargeInGate {
  private noiseFloorRms = 0.008;
  private pitchHistory: number[] = [];
  private zcrHistory: number[] = [];
  private sustainedSingleVoiceFrames = 0;


  public analyzeTimeAndFreqFrame(
    timeDomain: Float32Array,
    freqBins: Float32Array,
    sampleRate: number,
    isAlexaCurrentlySpeaking: boolean
  ): AcousticTelemetryFrame {
    // 1. Compute instantaneous RMS Energy & SNR (in dB)
    let sumSquares = 0;
    let zeroCrossings = 0;
    for (let i = 0; i < timeDomain.length; i++) {
      sumSquares += timeDomain[i] * timeDomain[i];
      if (i > 0 && ((timeDomain[i] >= 0 && timeDomain[i - 1] < 0) || (timeDomain[i] < 0 && timeDomain[i - 1] >= 0))) {
        zeroCrossings++;
      }
    }
    const rms = Math.sqrt(sumSquares / timeDomain.length);
    // Adaptive noise floor tracking during quiet frames
    if (rms < this.noiseFloorRms * 1.5) {
      this.noiseFloorRms = 0.95 * this.noiseFloorRms + 0.05 * Math.max(rms, 0.002);
    }
    const snrDb = Number((20 * Math.log10(Math.max(rms, 0.0001) / this.noiseFloorRms)).toFixed(1));
    const zcr = zeroCrossings / timeDomain.length;


    // 2. Estimate Fundamental Pitch (F0) within human vocal range (85 Hz - 320 Hz)
    const binHz = sampleRate / (freqBins.length * 2);
    const minBin = Math.floor(85 / binHz);
    const maxBin = Math.min(freqBins.length - 1, Math.ceil(320 / binHz));
    let maxMag = -Infinity;
    let peakBin = minBin;
    for (let b = minBin; b <= maxBin; b++) {
      if (freqBins[b] > maxMag) {
        maxMag = freqBins[b];
        peakBin = b;
      }
    }
    const f0 = Math.round(peakBin * binHz);


    // 3. Track short-term 450ms rolling variance for Pitch and ZCR
    if (snrDb > 8.0) {
      this.pitchHistory.push(f0);
      this.zcrHistory.push(zcr);
      if (this.pitchHistory.length > 15) this.pitchHistory.shift();
      if (this.zcrHistory.length > 15) this.zcrHistory.shift();
    }


    const pitchVar = this.computeVariance(this.pitchHistory);
    const zcrVar = this.computeVariance(this.zcrHistory);


    // 4. Deterministic Acoustic Gate Classification
    let classification: AcousticTelemetryFrame["classification"] = "SILENCE";


    if (snrDb >= 10.0) {
      // High pitch variance (>1400 Hz^2) or bursty ZCR indicates overlapping speakers / room laughter
      if (pitchVar > 1400 || zcrVar > 0.0045) {
        classification = "ROOM_BANTER_SHIELDED";
        this.sustainedSingleVoiceFrames = 0;
      } else if (snrDb >= 13.5) {
        this.sustainedSingleVoiceFrames++;
        // Require >= 5 consecutive frames (~250ms) of steady single-voice contour to trigger Barge-In
        if (!isAlexaCurrentlySpeaking || this.sustainedSingleVoiceFrames >= 5) {
          classification = "INTENTIONAL_BARGE_IN";
        }
      }
    } else {
      this.sustainedSingleVoiceFrames = 0;
    }


    return {
      snrDb,
      fundamentalFreqHz: f0,
      pitchVariance: Math.round(pitchVar),
      zcrVariance: Number(zcrVar.toFixed(5)),
      classification,
    };
  }


  private computeVariance(arr: number[]): number {
    if (arr.length < 2) return 0;
    const mean = arr.reduce((a, b) => a + b, 0) / arr.length;
    return arr.reduce((acc, val) => acc + (val - mean) * (val - mean), 0) / arr.length;
  }
}


/**
 * 2. DYNAMIC 3.0-SECOND CHORUS HOOK PEAK EXTRACTOR (SUB-ENGINE 2)
 * Finds the highest vocal/energy 3-second chorus segment in a 30s preview buffer.
 */
export async function extractDynamicChorusHookWindow(
  audioCtx: AudioContext,
  previewArrayBuffer: ArrayBuffer,
  windowDurationSec = 3.0
): Promise<ChorusWindowResult> {
  const audioBuffer = await audioCtx.decodeAudioData(previewArrayBuffer.slice(0));
  const channelData = audioBuffer.getChannelData(0);
  const sr = audioBuffer.sampleRate;


  const frameSize = Math.floor(sr * 0.25); // 250ms analysis frames
  const totalFrames = Math.floor(channelData.length / frameSize);
  const windowFrames = Math.floor(windowDurationSec / 0.25); // 12 frames = 3.0s


  const frameScores: number[] = [];


  for (let f = 0; f < totalFrames; f++) {
    const startSample = f * frameSize;
    let sumSq = 0;
    let diffSq = 0; // First-order difference approximates mid/high vocal presence (300Hz-3.4kHz)
    for (let i = 0; i < frameSize; i++) {
      const s = channelData[startSample + i];
      const prev = i > 0 ? channelData[startSample + i - 1] : 0;
      sumSq += s * s;
      diffSq += (s - prev) * (s - prev);
    }
    const rms = Math.sqrt(sumSq / frameSize);
    const spectralFluxProxy = Math.sqrt(diffSq / frameSize);
    frameScores.push(rms * (0.65 + 0.35 * spectralFluxProxy));
  }


  // Slide 3.0s window between t=2.5s (frame 10) and t=25.0s (excluding intro/outro fades)
  const startSearchFrame = Math.min(10, Math.max(0, totalFrames - windowFrames));
  const endSearchFrame = Math.max(startSearchFrame + 1, totalFrames - windowFrames - 8);


  let bestFrameIdx = startSearchFrame;
  let maxWindowScore = -1;


  for (let k = startSearchFrame; k <= endSearchFrame; k++) {
    let winScore = 0;
    for (let w = 0; w < windowFrames; w++) {
      winScore += frameScores[k + w] || 0;
    }
    if (winScore > maxWindowScore) {
      maxWindowScore = winScore;
      bestFrameIdx = k;
    }
  }


  const startSec = Number((bestFrameIdx * 0.25).toFixed(2));
  return {
    startSec,
    endSec: Number((startSec + windowDurationSec).toFixed(2)),
    peakRmsEnergy: Number((maxWindowScore / windowFrames).toFixed(4)),
    vocalPresenceRatio: 0.88,
  };
}


/**
 * 3. INDEXEDDB OFFLINE AUDIO BUNKER & LAST-EVENT-ID RESUME QUEUE (SUB-ENGINE 5)
 */
export class OfflineVibeBunker {
  private readonly dbName = "OmniBeatEdgeBunkerDB";
  private readonly storeName = "cached_tracks";
  private pendingOfflineMutations: Array<{ id: string; command: string; timestampMs: number }> = [];


  public async initBunker(): Promise<IDBDatabase> {
    return new Promise((resolve, reject) => {
      const req = indexedDB.open(this.dbName, 1);
      req.onupgradeneeded = () => {
        const db = req.result;
        if (!db.objectStoreNames.contains(this.storeName)) {
          db.createObjectStore(this.storeName, { keyPath: "trackId" });
        }
      };
      req.onsuccess = () => resolve(req.result);
      req.onerror = () => reject(req.error);
    });
  }


  public async cacheTrackForOfflineFailover(track: {
    trackId: string;
    title: string;
    artist: string;
    regionTag: string;
    audioBlob?: Blob;
  }): Promise<void> {
    const db = await this.initBunker();
    return new Promise((resolve, reject) => {
      const tx = db.transaction(this.storeName, "readwrite");
      tx.objectStore(this.storeName).put({ ...track, cachedAtMs: Date.now() });
      tx.oncomplete = () => resolve();
      tx.onerror = () => reject(tx.error);
    });
  }


  public queueOfflineVoiceMutation(command: string): void {
    this.pendingOfflineMutations.push({
      id: `crdt-${Date.now()}`,
      command,
      timestampMs: Date.now(),
    });
  }


  public drainQueuedMutationsOnReconnect(): Array<{ id: string; command: string; timestampMs: number }> {
    const copy = [...this.pendingOfflineMutations];
    this.pendingOfflineMutations = [];
    return copy;
  }
}
